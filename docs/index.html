Meet e# (esharp)

A Rust-inspired programming language for combining Steam games together into a single playable experience.

use api_board::{Movie, Weather, Steam};

game! SteamMashup {
    title: "Cinematic weather in Berlin",
    scenes: [
        Scene::from(Movie::search("inception")),
        Scene::from(Weather::current("Berlin")),
        Scene::from(Steam::library().pick("Portal 2")),
    ],
}

fn main() -> Game {
    let mut game = SteamMashup::new();
    game.play()
}
